from datasets import load_dataset
import pandas as pd
import re
from collections import Counter

def load_medqa_dataset(subset_size=None):
    """
    Load the MedQA-USMLE dataset
    
    Args:
        subset_size: If provided, limit to this many questions
        
    Returns:
        Dataset object
    """
    dataset = load_dataset("GBaker/MedQA-USMLE-4-options")
    
    # Convert to pandas for easier analysis
    train_df = pd.DataFrame(dataset["train"])
    
    # Apply subset limit if specified
    if subset_size and subset_size < len(train_df):
        train_df = train_df.head(subset_size)
    
    return train_df

def analyze_questions(df):
    """
    Analyze questions to identify categories
    
    Args:
        df: DataFrame with questions
        
    Returns:
        Dictionary with analysis results
    """
    # Extract potential categories based on medical terms
    categories = []
    for q in df['question']:
        # Extract medical specialties mentioned in questions
        if re.search(r'cardio|heart|arter|vein|circulat', q, re.I):
            categories.append('cardiology')
        elif re.search(r'neuro|brain|spinal|nerve', q, re.I):
            categories.append('neurology')
        elif re.search(r'gastro|stomach|intestine|liver', q, re.I):
            categories.append('gastroenterology')
        # Add more category patterns as needed
        else:
            categories.append('other')
            
    # Count categories
    category_counts = Counter(categories)
    
    # Analyze answer distribution
    answer_distribution = df['answer_idx'].value_counts().to_dict()
    
    # Analyze question length
    df['question_length'] = df['question'].apply(len)
    
    return {
        'category_counts': category_counts,
        'answer_distribution': answer_distribution,
        'avg_question_length': df['question_length'].mean(),
        'min_question_length': df['question_length'].min(),
        'max_question_length': df['question_length'].max(),
    }

import wikipediaapi
import wikipedia
import time
import re
from tqdm import tqdm

class WikiSearcher:
    def __init__(self, user_agent='MedicalQABot/1.0'):
        """
        Initialize Wikipedia API
        
        Args:
            user_agent: Custom user agent for API requests
        """
        self.wiki_api = wikipediaapi.Wikipedia(
            language='en',
            extract_format=wikipediaapi.ExtractFormat.WIKI,
            user_agent=user_agent
        )
        
        # Set rate limits to be respectful to Wikipedia servers
        self.request_delay = 1  # seconds
        
    def search_wikipedia(self, query, max_results=5):
        """
        Search Wikipedia for relevant pages
        
        Args:
            query: Search query
            max_results: Maximum number of results to return
            
        Returns:
            List of page titles
        """
        try:
            # Respect rate limits
            time.sleep(self.request_delay)
            
            # Search Wikipedia
            results = wikipedia.search(query, results=max_results)
            return results
        except Exception as e:
            print(f"Error searching Wikipedia: {e}")
            return []
    
    def get_page_content(self, page_title):
        """
        Get content of a Wikipedia page
        
        Args:
            page_title: Title of the page
            
        Returns:
            Text content of the page or None if not found
        """
        try:
            # Respect rate limits
            time.sleep(self.request_delay)
            
            page = self.wiki_api.page(page_title)
            if page.exists():
                return page.text
            return None
        except Exception as e:
            print(f"Error retrieving page {page_title}: {e}")
            return None
            
    def search_for_question(self, question, options, max_pages=5):
        """
        Search Wikipedia for pages relevant to the question and options
        
        Args:
            question: Question text
            options: List of answer options
            max_pages: Maximum number of pages to retrieve
            
        Returns:
            Dictionary mapping page titles to their content
        """
        # Extract keywords from question
        keywords = self._extract_keywords(question)
        
        # Create search queries from question and each option
        queries = [question] + [f"{question} {opt}" for opt in options]
        
        # Track pages we've already retrieved
        retrieved_pages = {}
        
        # For each query, search until we have max_pages or run out of queries
        for query in queries:
            if len(retrieved_pages) >= max_pages:
                break
                
            # Get search results
            page_titles = self.search_wikipedia(query)
            
            # Retrieve content for each page
            for title in page_titles:
                if len(retrieved_pages) >= max_pages:
                    break
                if title not in retrieved_pages:
                    content = self.get_page_content(title)
                    if content:
                        retrieved_pages[title] = content
        
        return retrieved_pages
        
    def _extract_keywords(self, text):
        """
        Extract important keywords from text
        
        Args:
            text: Input text
            
        Returns:
            List of keywords
        """
        # Remove common words and keep mainly medical terms
        # This is a very simple approach - could be improved with NLP techniques
        words = re.findall(r'\b[A-Za-z]{3,}\b', text)
        stopwords = {'the', 'and', 'that', 'this', 'with', 'from', 'what', 'which'}
        keywords = [word for word in words if word.lower() not in stopwords]
        return keywords

from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np
import re
import nltk
from nltk.corpus import stopwords

# Download NLTK resources
try:
    nltk.data.find('corpora/stopwords')
except LookupError:
    nltk.download('stopwords')

class TfidfQuestionAnswerer:
    def __init__(self):
        """Initialize the TF-IDF based question answerer model"""
        self.vectorizer = TfidfVectorizer(
            stop_words='english', 
            ngram_range=(1, 2),
            max_features=10000
        )
        self.stop_words = set(stopwords.words('english'))
        
    def preprocess_text(self, text):
        """Clean and preprocess text"""
        # Convert to lowercase
        text = text.lower()
        # Remove special characters
        text = re.sub(r'[^\w\s]', ' ', text)
        # Remove numbers
        text = re.sub(r'\d+', ' ', text)
        # Remove extra whitespace
        text = re.sub(r'\s+', ' ', text).strip()
        return text
        
    def answer_question(self, question, options, wiki_pages):
        """
        Answer medical question using Wikipedia pages
        
        Args:
            question: Question text
            options: List of answer options
            wiki_pages: Dictionary of Wiki page titles to content
            
        Returns:
            index of predicted answer, confidence scores
        """
        # Combine all wiki content
        combined_content = " ".join(wiki_pages.values())
        
        # Preprocess
        processed_content = self.preprocess_text(combined_content)
        processed_question = self.preprocess_text(question)
        processed_options = [self.preprocess_text(opt) for opt in options]
        
        # Create option-specific queries
        option_queries = [f"{processed_question} {opt}" for opt in processed_options]
        
        # Fit vectorizer on content
        self.vectorizer.fit([processed_content])
        
        # Transform content and queries
        content_vector = self.vectorizer.transform([processed_content])
        query_vectors = self.vectorizer.transform(option_queries)
        
        # Calculate similarity scores
        similarity_scores = cosine_similarity(query_vectors, content_vector).flatten()
        
        # Get predicted answer index
        predicted_idx = np.argmax(similarity_scores)
        
        return predicted_idx, similarity_scores

class EnsembleQuestionAnswerer:
    """
    Enhanced question answerer that uses multiple strategies
    """
    def __init__(self):
        self.tfidf_model = TfidfQuestionAnswerer()
        
    def answer_question(self, question, options, wiki_pages):
        """
        Answer using ensemble of methods
        
        Args:
            question: Question text
            options: List of answer options
            wiki_pages: Dictionary of Wiki page titles to content
            
        Returns:
            index of predicted answer, confidence scores
        """
        # For now, just use TFIDF model
        # Later we can add other models and combine their predictions
        return self.tfidf_model.answer_question(question, options, wiki_pages)


import argparse

def main():
    # Parse command line arguments
    parser = argparse.ArgumentParser(description='Medical Question Answering System')
    parser.add_argument('--subset', type=int, default=None,
                        help='Number of questions to evaluate (default: all)')
    parser.add_argument('--analyze', action='store_true', 
                        help='Perform dataset analysis only')
    parser.add_argument('--verbose', action='store_true',
                        help='Print detailed output during evaluation')
    args = parser.parse_args()
    
    # Load the dataset
    print(f"Loading dataset{' (subset: ' + str(args.subset) + ')' if args.subset else ''}...")
    dataset = load_medqa_dataset(subset_size=args.subset)
    print(f"Loaded {len(dataset)} questions.")
    
    # If analyze flag is set, perform analysis and exit
    if args.analyze:
        print("Analyzing dataset...")
        analysis = analyze_questions(dataset)
        print("\nAnalysis Results:")
        for key, value in analysis.items():
            print(f"{key}: {value}")
        return
    
    # Initialize components
    wiki_searcher = WikiSearcher()
    qa_model = EnsembleQuestionAnswerer()
    
    # Process questions
    correct_count = 0
    
    print("Starting question answering...")
    for i, row in tqdm(dataset.iterrows(), total=len(dataset)):
        question = row['question']
        options = [row['options']['A'], row['options']['B'], 
                  row['options']['C'], row['options']['D']]
        correct_idx = row['answer_idx']
        
        if args.verbose:
            print(f"\nQuestion {i+1}: {question}")
            print(f"Options: {options}")
            print(f"Correct answer: {options[correct_idx]} (index: {correct_idx})")
        
        # Search Wikipedia for relevant pages
        wiki_pages = wiki_searcher.search_for_question(question, options)
        
        if args.verbose:
            print(f"Retrieved {len(wiki_pages)} Wikipedia pages")
        
        # Answer the question
        predicted_idx, confidence_scores = qa_model.answer_question(question, options, wiki_pages)
        
        # Check if prediction is correct
        is_correct = (predicted_idx == correct_idx)
        if is_correct:
            correct_count += 1
            
        if args.verbose:
            print(f"Predicted answer: {options[predicted_idx]} (index: {predicted_idx})")
            print(f"Confidence scores: {confidence_scores}")
            print(f"Correct: {is_correct}")
    
    # Calculate accuracy
    accuracy = correct_count / len(dataset)
    print(f"\nEvaluation Results:")
    print(f"Total questions: {len(dataset)}")
    print(f"Correctly answered: {correct_count}")
    print(f"Accuracy: {accuracy:.4f} ({correct_count}/{len(dataset)})")

if __name__ == "__main__":
    main()
